   branches: [ main, develop ]

 validate-schema:
   runs-on: ubuntu-latest
   steps:
     - name: Checkout Repository Source
       uses: actions/checkout@v4

     - name: Setup Node.js Environment
       uses: actions/setup-node@v4
       with:
         node-version: '20'

     - name: Validate Gateway JSON Schema
       run: |
         npx ajv-cli validate -s config/gateway-policies.json -d config/gateway-policies.json

  integrity-check:
   runs-on: ubuntu-latest
